\documentclass[titlepage]{generalreport} % titlepage: title on its own page,
                                         % numbering starts on page 2

% ---- Title page head --------------------------------------------------------

                                       % right=.333, bottom=.333]

                                       % top left, free text; wraps at half the
                                       % text width, \\ breaks manually

% ---- Title block -----------------------------------------------------------
\documenttype{BERICHT}                 % label above the title, as typed
